def handle_client(client_socket):

    ssh_socket.settimeout(15)
    try:
        ssh_socket.connect(('127.0.0.1', 109))

            data = client_socket.recv(1)

            try:
                while True:
                    buffer = src.recv(8192)
                    if not buffer: break
                    dst.send(buffer)
            except: pass
            finally: 
                src.close()
                dst.close()

cat <<'EOF' > /usr/local/bin/ws-proxy.py
import socket, threading

def handle_client(client_socket):
    client_socket.settimeout(15)
    ssh_socket = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    ssh_socket.settimeout(15)
    try:
        ssh_socket.connect(('127.0.0.1', 109))
        req = b""
        while True:
            data = client_socket.recv(1)
            if not data: break
            req += data
            if b"\r\n\r\n" in req: break
            
        client_socket.send(b"HTTP/1.1 101 Switching Protocols\r\nUpgrade: websocket\r\nConnection: Upgrade\r\n\r\n")
        
        client_socket.settimeout(None)
        ssh_socket.settimeout(None)
        
        def forward(src, dst):
            try:
                while True:
                    buffer = src.recv(8192)
                    if not buffer: break
                    dst.send(buffer)
            except: pass
            finally: 
                src.close()
                dst.close()
                
        threading.Thread(target=forward, args=(client_socket, ssh_socket)).start()
        threading.Thread(target=forward, args=(ssh_socket, client_socket)).start()
    except: 
        client_socket.close()
        ssh_socket.close()

server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
server.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
server.bind(('127.0.0.1', 8880))
server.listen(5000)
while True:
    try:
        client_socket, addr = server.accept()
        threading.Thread(target=handle_client, args=(client_socket,)).start()
    except: pass
EOF

# Create Bash Auth Script for Multi-User Support (Unbuffered, Sandbox-Safe)
cat << 'EOF' > /etc/hysteria/auth.sh
#!/bin/bash
# Grab all input instantly without waiting for a newline
INPUT=$(cat)

# Extract the password the user typed
PASS=$(echo "$INPUT" | jq -r '.auth' 2>/dev/null)

if [ -z "$PASS" ] || [ "$PASS" == "null" ]; then
    echo "false"
    exit 0
fi

# Scrub the database of any invisible formatting (\r) that causes false rejections
sed -i 's/\r//g' /etc/falcon/data/hy2_users.txt

# Verify the password
if awk '{print $2}' /etc/falcon/data/hy2_users.txt 2>/dev/null | grep -q -x "$PASS"; then
    echo "true"
else
    echo "false"
fi
EOF

chmod +x /etc/hysteria/auth.sh
